handleStartTrigger()

الزر الفيزيائي: بدء مؤجّل 0.5 ثانية + كشف الحافّتين.

① البدء المؤجّل

if (startScheduled && !isRunning && !manualMode
    && millis() - startScheduledTime >= 500) {
  startScheduled = false;
  startRobot();
}

يعطي فرصة لإبعاد اليد قبل انطلاق الروبوت.

② كشف الضغط والإفلات

// ضغط (Falling edge)
if (digitalRead(BUTTON_PIN) == LOW && !buttonPressed) {
  buttonPressed = true;
  currentTriggerMode = "button";
  if (isRunning)      stopRobot();
  else if (!manualMode) startScheduled = true;
}
// إفلات (Rising edge)
if (digitalRead(BUTTON_PIN) == HIGH && buttonPressed)
  buttonPressed = false;
Toggle: الزر الواحد يبدأ ويوقف. في الوضع اليدوي لا يبدأ (!manualMode)، لكنه يوقف لو شغّال.

FollowPathPID()

قلب المشروع: خطأ مرجّح ← PID ← سرعتان ← تصحيح.

① الأوزان (5 حساسات)

int weights[5] = { -1, -2, 0, 2, 1 };
الحساس01234
الوزن−1−20+2+1

الحساسات الداخلية أقوى → تصحيح أسرع قرب المنتصف.

② حساب الخطأ

count = 0;
for (int i = 0; i < 5; i++) {
  if (sensorReadings[i] == HIGH) {
    error += weights[i];
    count++;
  }
}
  • error > 0 → الخط يمين → يلف يمين
  • error < 0 → الخط شمال → يلف شمال
  • error = 0 → في المنتصف
  • count = 0 → لا خط ظاهر

③ التسلسل

readSensors()
↓
حساب error / count
↓
PID → correction
↓
speedL / speedR
↓
count == 5 ؟
↓ نعم
move(STOP) → endProcedure()
↓ لا
setMotorSpeeds() → detect90()

④ الحمايات

  • if (manualMode) return; — تعطيل كامل.
  • if (isRunning) — لا عمل إن لم تبدأ الجولة.
  • count == 5 — كل الحساسات على الأسود = نهاية المسار.
  • return بعد endProcedure() — خروج فوري.
14

حساب PID

ثلاثة حدود: تناسبي + تكاملي + تفاضلي.

الكود

integral += error;
integral = constrain(integral, -Imax, Imax);   // ±50

float derivative = error - lastError;

correction = Kp * error
           + Ki * integral
           + Kd * derivative;

lastError = error;

constrain

الحدود الثلاثة

الحدالمعنىالتأثير
P = Kp·eحجم الانحراف الآنرد فعل فوري
I = Ki·Σeتراكم الانحرافيزيل الخطأ المستمر
D = Kd·Δeسرعة تغيّر الانحرافيخفّف التذبذب

القيم الافتراضية: Kp=35 · Ki=0.05 · Kd=25

Windup: حدّ Imax = 50 يمنع تضخّم الحد التكاملي عند بقاء الخطأ في اتجاه واحد طويلاً.

تحويل التصحيح إلى سرعات

التصحيح يُضاف لعجلة ويُطرح من الأخرى.

الكود

int base = 255 * speedPercentage / 100;

int speedL = constrain(base - correction, -255, 255);
int speedR = constrain(base + correction, -255, 255);

المنطق

detect90()

كشف المنعطف الحاد عند انتهاء الخط تحت الحساس الأوسط.

منطق الكشف

خطوات ما بعد الكشف

18

turnRightdetection() / turnLeftdetection()

مسح دوراني بحثاً عن الخط بعد المنعطف.

يمين — turnRightdetection()

move(STOP); delay(50);
while (sensorReadings[3] == LOW
    && sensorReadings[4] == LOW) {
  setMotorSpeeds(100, 0);
  readSensors();
  if (sensorReadings[3] == HIGH
   || sensorReadings[4] == HIGH) break;
  delay(10);
}

يدور حتى يرى الحساسان 3 أو 4 الخط.

شمال — turnLeftdetection()

move(STOP); delay(50);
while (sensorReadings[0] == LOW
    && sensorReadings[1] == LOW) {
  setMotorSpeeds(0, 100);
  readSensors();
  if (sensorReadings[0] == HIGH
   || sensorReadings[1] == HIGH) break;
  delay(10);
}

يدور حتى يرى الحساسان 0 أو 1 الخط.

ملاحظة

السرعة ثابتة 100 ولا تعتمد على speedPercentage — مسح بطيء ومضبوط.

⚠️

لو الخط غير موجود → حلقة لا نهائية (لا timeout). مسجّل كملاحظة #2.

إنهاء الجولة عند الوصول لنهاية المسار.

الخطوات الست

↓

20

updateStopwatch()

عدّاد زمني يُحدَّث كل 50ms أثناء التشغيل.

الكود

void updateStopwatch() {
  unsigned long now = millis();
  if (now - lastStopwatchUpdate >= 50) {
    stopwatchValue = now - startTime;
    lastStopwatchUpdate = now;
  }
}

الفاصل: stopwatchUpdateInterval = 50.

لماذا لا نحدّث كل دورة؟

  • تقليل العمليات وضغط الـ JSON.
  • دقة 50ms كافية للعرض (سنتيمتر من الثانية = 10ms تقريباً).
  • قيمة العداد تُقرأ بعد ذلك في /status وaddRoundToHistory.
⚠️ تُنادى فقط عند !manualMode && isRunning — لذلك يتوقف العدّاد عند الإيقاف، لكنه لا يُصفَّر إلا في startRobot()/clearHistory().

addRoundToHistory() / clearHistory()

سجل آخر 10 جولات — الأحدث أولاً.

الإضافة

if (stopwatchValue == 0 && status != "Finished 🏁") return;

if (historyCount >= 10) {
  for (int i = 9; i > 0; i--)
    roundHistory[i] = roundHistory[i-1];
} else {
  for (int i = historyCount; i > 0; i--)
    roundHistory[i] = roundHistory[i-1];
  historyCount++;
}
roundHistory[0] = newRecord;
currentRound++;

السلوك

  • السجل محدود بـ 10 — الأقدم يُحذف.
  • الإزاحة تجعل [0] أحدث جولة.
  • تُتجاهل الجولة لو stopwatchValue == 0 (لم تبدأ فعلياً).
  • currentRound++ بعد كل تسجيل.

clearHistory()

  • historyCount = 0 · currentRound = 1.
  • تصفير Trigger modes + stopwatch.
  • robotStatus = "Ready".

المسارالاستدعاءيدوي فقط؟ملاحظة
/forward · /backwardmove(FORWARD/BACKWARD)✅يستمر حتى أمر آخر
/left · /rightmove(LEFT/RIGHT)✅لف بعجلة واحدة
/cw · /ccwmove(ROTATE_*)✅دوران مستمر
/stopmove(STOP)❌ أي وضع⚠️ لحظي — لا يوقف isRunning
/turnRight90 · /turnLeft90لف ثابت 130 PWM✅⚠️ بدون رد لو مش يدوي
/horn?state=on|offBUZZER مباشرة❌غير حاجز
/clearhistoryclearHistory()❌POST فقط

الواجهة اليدوية

  • ضغط مستمر → تكرار الأمر كل 100ms.
  • إفلات → /stop.
  • وضع 90° → أمر واحد لكل ضغطة.

اللف 90°

  • move(STOP) → delay(50)
  • setMotorSpeeds(±130) → delay(turnDelay)
  • readSensors() → توقف
  • ⏱️ حاجز ~350ms
